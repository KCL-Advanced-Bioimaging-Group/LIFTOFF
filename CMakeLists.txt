cmake_minimum_required(VERSION 3.20)
project(LIFTOFFNative VERSION 3.3.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(LeicaFlimDecoder SHARED native/leica_flim_decoder.cpp native/leica_raw_decoder.cpp)
add_library(LeicaLmsrawFilter SHARED native/leica_lmsraw_filter.cpp)
foreach(target LeicaFlimDecoder LeicaLmsrawFilter)
  target_link_libraries(${target} PRIVATE Threads::Threads)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /O2 /EHsc)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -O3)
  endif()
endforeach()
install(TARGETS LeicaFlimDecoder LeicaLmsrawFilter
  RUNTIME DESTINATION leica_flim_converter/native
  LIBRARY DESTINATION leica_flim_converter/native)
